// 规划阶段夹具：隔离 Redis 中清除脚本缓存，验证仍有活动会话。
func TestS07PlanSessionBatchAfterScriptFlush(t *testing.T){ctx:=context.Background();rdb:=testRedis(t);c:=NewSessionLimitCache(rdb,5);ok,err:=c.RegisterSession(ctx,91001,"planning-session",2,time.Minute);require.NoError(t,err);require.True(t,ok);require.NoError(t,rdb.ScriptFlush(ctx).Err());got,err:=c.GetActiveSessionCountBatch(ctx,[]int64{91001},nil);require.NoError(t,err);require.Equal(t,1,got[91001],"脚本缓存丢失后仍应读到活动会话")}
// 规划阶段夹具：序列号较小的事务在水位推进后才提交。
func TestS07PlanOutboxLateCommit(t *testing.T){ctx:=context.Background();_,err:=integrationDB.ExecContext(ctx,"TRUNCATE scheduler_outbox");require.NoError(t,err);tx,err:=integrationDB.BeginTx(ctx,nil);require.NoError(t,err);defer tx.Rollback();var low,high int64;require.NoError(t,tx.QueryRowContext(ctx,"INSERT INTO scheduler_outbox(event_type) VALUES($1) RETURNING id",service.SchedulerOutboxEventFullRebuild).Scan(&low));require.NoError(t,integrationDB.QueryRowContext(ctx,"INSERT INTO scheduler_outbox(event_type) VALUES($1) RETURNING id",service.SchedulerOutboxEventFullRebuild).Scan(&high));r:=NewSchedulerOutboxRepository(integrationDB);events,err:=r.ListAfterAndReleaseDedup(ctx,0,200);require.NoError(t,err);require.Len(t,events,1);require.Equal(t,high,events[0].ID);require.NoError(t,tx.Commit());later,err:=r.ListAfterAndReleaseDedup(ctx,high,200);require.NoError(t,err);found:=false;for _,e:=range later{if e.ID==low{found=true}};require.True(t,found,"较小 ID 迟提交后被水位跳过")}

// 规划阶段夹具：旧 bucket 锁过期后，新持有者不应被旧释放删除。
func TestS07PlanBucketExpiredOwnerRelease(t *testing.T){ctx:=context.Background();rdb:=testRedis(t);a:=NewSchedulerCache(rdb);b:=NewSchedulerCache(rdb);bucket:=service.SchedulerBucket{GroupID:91001,Platform:service.PlatformOpenAI,Mode:service.SchedulerModeSingle};ok,err:=a.TryLockBucket(ctx,bucket,20*time.Millisecond);require.NoError(t,err);require.True(t,ok);time.Sleep(40*time.Millisecond);ok,err=b.TryLockBucket(ctx,bucket,time.Minute);require.NoError(t,err);require.True(t,ok);require.NoError(t,a.UnlockBucket(ctx,bucket));ok,err=a.TryLockBucket(ctx,bucket,time.Minute);require.NoError(t,err);require.False(t,ok,"旧持有者释放不应删除仍有效的新锁")}
