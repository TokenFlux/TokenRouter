// 仅使用本测试进程的隔离 PostgreSQL，触发器模拟留痕落库失败。
func TestS08PlanningAuditClearRollbackPostgres(t *testing.T){
 ctx:=context.Background();repo:=NewAuditLogRepository(integrationDB);s:=service.NewAuditLogService(repo,nil);defer s.Stop()
 require.NoError(t,repo.Insert(ctx,&service.AuditLog{Action:"planning-original",CreatedAt:time.Now()}))
 _,e:=integrationDB.ExecContext(ctx,`CREATE FUNCTION s08_reject_clear_trace() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN IF NEW.action='admin.audit_log.clear' THEN RAISE EXCEPTION 'planning trace failure'; END IF; RETURN NEW; END $$; CREATE TRIGGER s08_reject_clear_trace BEFORE INSERT ON audit_logs FOR EACH ROW EXECUTE FUNCTION s08_reject_clear_trace()`);require.NoError(t,e)
 defer integrationDB.ExecContext(ctx,`DROP TRIGGER s08_reject_clear_trace ON audit_logs; DROP FUNCTION s08_reject_clear_trace()`)
 deleted,e:=s.ClearAll(ctx,&service.AuditLog{});require.Error(t,e);t.Logf("deleted=%d error=%v",deleted,e)
 n,e:=repo.Count(ctx);require.NoError(t,e);require.Equal(t,int64(1),n,"failed mandatory trace must retain original audit history")
}
type s08PausedPostgresState struct{*dashboardAggregationRepository;reads atomic.Int32;entered chan struct{};release chan struct{}}
func(r *s08PausedPostgresState)GetUsageAnalyticsAggregationState(ctx context.Context)(*service.UsageAnalyticsAggregationState,error){v,e:=r.dashboardAggregationRepository.GetUsageAnalyticsAggregationState(ctx);if r.reads.Add(1)==1{close(r.entered);<-r.release};return v,e}
func TestS08PlanningBackfillWatermarkPostgres(t *testing.T){
 ctx:=context.Background();base:=newDashboardAggregationRepositoryWithSQL(integrationDB);old:=time.Date(2026,9,1,1,0,0,0,time.UTC);next:=old.Add(time.Hour);require.NoError(t,base.SaveUsageAnalyticsAggregationState(ctx,&service.UsageAnalyticsAggregationState{LiveWatermark:old,Phase:"idle"}))
 r:=&s08PausedPostgresState{dashboardAggregationRepository:base,entered:make(chan struct{}),release:make(chan struct{})};s:=service.NewDashboardAggregationService(r,nil,&config.Config{DashboardAgg:config.DashboardAggregationConfig{BackfillEnabled:true}});s.SetPreAggregationSettings(nil)
 done:=make(chan error,1);go func(){done<-s.TriggerBackfill(old.Add(-time.Hour),old)}();<-r.entered
 require.NoError(t,base.SaveUsageAnalyticsAggregationState(ctx,&service.UsageAnalyticsAggregationState{LiveWatermark:next,Phase:"idle"}));close(r.release);require.NoError(t,<-done)
 got,e:=base.GetUsageAnalyticsAggregationState(ctx);require.NoError(t,e);require.True(t,got.LiveWatermark.Equal(next),"manual backfill restored stale watermark: %v",got.LiveWatermark)
}

// 现有存储端口的计数器用于固定 SQL 查询次数与小型夹具 Explain 基线。
type s08SQLCall struct{ Query string `json:"query"`;Args []any `json:"-"` }
type s08CountingSQL struct{sqlExecutor;calls []s08SQLCall}
func(s *s08CountingSQL)QueryContext(ctx context.Context,q string,a ...any)(*sql.Rows,error){s.calls=append(s.calls,s08SQLCall{q,a});return s.sqlExecutor.QueryContext(ctx,q,a...)}
func TestS08PlanningQueryShapeBaseline(t *testing.T){
 ctx:=context.Background();tx:=testEntTx(t);client:=tx.Client();writer:=newUsageLogRepositoryWithSQL(client,tx);account:=mustCreateAccount(t,client,&service.Account{Name:"s08-query-shape"});ids:=[]int64{};keys:=[]int64{}
 for i:=0;i<8;i++{u:=mustCreateUser(t,client,&service.User{Email:fmt.Sprintf("s08-query-%d@test.local",i)});key:=mustCreateApiKey(t,client,&service.APIKey{UserID:u.ID,Key:fmt.Sprintf("sk-s08-query-%d",i),Name:"k"});ids=append(ids,u.ID);keys=append(keys,key.ID);for j:=0;j<4;j++{_,e:=writer.Create(ctx,&service.UsageLog{UserID:u.ID,APIKeyID:key.ID,AccountID:account.ID,Model:"planning",InputTokens:10,OutputTokens:5,TotalCost:0.1,ActualCost:0.1,CreatedAt:time.Now().Add(-time.Hour)});require.NoError(t,e)}}
 counter:=&s08CountingSQL{sqlExecutor:tx};repo:=newUsageLogRepositoryWithSQL(client,counter);start,end:=time.Now().Add(-2*time.Hour),time.Now();results:=map[string]any{}
 for _,n:=range []int{4,8}{counter.calls=nil;_,e:=repo.GetBatchUserUsageStats(ctx,ids[:n],start,end);require.NoError(t,e);results[fmt.Sprintf("batch_users_%d",n)]=len(counter.calls);counter.calls=nil;_,e=repo.GetBatchAPIKeyUsageStats(ctx,keys[:n],start,end);require.NoError(t,e);results[fmt.Sprintf("batch_keys_%d",n)]=len(counter.calls)}
 counter.calls=nil;_,e:=repo.GetUserSpendingRanking(ctx,start,end,8);require.NoError(t,e);results["user_ranking"]=len(counter.calls)
 plans:=[]any{};for _,call:=range counter.calls{rows,e:=tx.QueryContext(ctx,"EXPLAIN (FORMAT JSON) "+call.Query,call.Args...);require.NoError(t,e);for rows.Next(){var raw []byte;require.NoError(t,rows.Scan(&raw));plans=append(plans,json.RawMessage(raw))};require.NoError(t,rows.Close())};results["ranking_explain"]=plans
 raw,e:=json.MarshalIndent(results,"","  ");require.NoError(t,e);require.NoError(t,os.WriteFile("/tmp/tokenrouter-s08-planning/query-shape.json",raw,0600));t.Logf("counts=%s",raw)
}
type s08CancelCleanupPostgres struct{service.UsageCleanupRepository;taskID int64;operator int64;observed chan struct{};once atomic.Bool}
func(r *s08CancelCleanupPostgres)CreateTask(ctx context.Context,t *service.UsageCleanupTask)error{e:=r.UsageCleanupRepository.CreateTask(ctx,t);r.taskID=t.ID;return e}
func(r *s08CancelCleanupPostgres)DeleteUsageLogsBatch(ctx context.Context,f service.UsageCleanupFilters,n int)(int64,error){d,e:=r.UsageCleanupRepository.DeleteUsageLogsBatch(ctx,f,n);if e==nil && d>0 {_,e=r.UsageCleanupRepository.CancelTask(ctx,r.taskID,r.operator)};return d,e}
func(r *s08CancelCleanupPostgres)GetTaskStatus(ctx context.Context,id int64)(string,error){status,e:=r.UsageCleanupRepository.GetTaskStatus(ctx,id);if status==service.UsageCleanupStatusCanceled && r.once.CompareAndSwap(false,true){close(r.observed)};return status,e}
type s08RecomputePostgres struct{*dashboardAggregationRepository;calls atomic.Int64}
func(r *s08RecomputePostgres)RecomputeRange(ctx context.Context,a,b time.Time)error{r.calls.Add(1);return r.dashboardAggregationRepository.RecomputeRange(ctx,a,b)}
func TestS08PlanningPartialCancelPostgres(t *testing.T){
 ctx:=context.Background();client:=testEntClient(t);u:=mustCreateUser(t,client,&service.User{Email:"s08-cancel@test.local"});key:=mustCreateApiKey(t,client,&service.APIKey{UserID:u.ID,Key:"sk-s08-cancel",Name:"k"});acc:=mustCreateAccount(t,client,&service.Account{Name:"s08-cancel"});repo:=newUsageLogRepositoryWithSQL(client,integrationDB);defer repo.StopUsageBatchers();now:=time.Now().UTC().Add(-72*time.Hour)
 for i:=0;i<2;i++{_,e:=repo.Create(ctx,&service.UsageLog{UserID:u.ID,APIKeyID:key.ID,AccountID:acc.ID,Model:"planning",TotalCost:1,ActualCost:1,CreatedAt:now});require.NoError(t,e)}
 base:=newDashboardAggregationRepositoryWithSQL(integrationDB);require.NoError(t,base.AggregateUsageAnalyticsRange(ctx,now.Add(-time.Hour),now.Add(time.Hour)))
 ar:=&s08RecomputePostgres{dashboardAggregationRepository:base};agg:=service.NewDashboardAggregationService(ar,nil,nil);agg.Start();defer agg.Stop()
 cr:=&s08CancelCleanupPostgres{UsageCleanupRepository:NewUsageCleanupRepository(client,integrationDB),operator:u.ID,observed:make(chan struct{})};cleanup:=service.NewUsageCleanupService(cr,nil,agg,&config.Config{UsageCleanup:config.UsageCleanupConfig{Enabled:true,BatchSize:1}})
 _,e:=cleanup.CreateTask(ctx,service.UsageCleanupFilters{StartTime:now.Add(-time.Hour),EndTime:now.Add(time.Hour),UserID:&u.ID},u.ID);require.NoError(t,e)
 select{case <-cr.observed:case <-time.After(5*time.Second):t.Fatal("cancel not observed")};cleanup.Stop();agg.Stop()
 var raw,total int64;require.NoError(t,integrationDB.QueryRowContext(ctx,"SELECT COUNT(*) FROM usage_logs WHERE user_id=$1",u.ID).Scan(&raw));require.NoError(t,integrationDB.QueryRowContext(ctx,"SELECT COALESCE(SUM(total_requests),0) FROM usage_analytics_hourly WHERE user_id=$1",u.ID).Scan(&total));t.Logf("raw=%d aggregate=%d recomputes=%d",raw,total,ar.calls.Load());require.Equal(t,raw,total,"canceled cleanup left stale analytic count")
}
